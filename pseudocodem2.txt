PROGRAM Simulasi_Osilator_Harmonik

// DEKLARASI KONSTANTA FISIKA
    omega = 2.0          // Frekuensi sudut
    x0 = 1.0             // Posisi awal
    v0 = 0.0             // Kecepatan awal
    T = 5.0              // Waktu total simulasi

// DEKLARASI VARIASI LANGKAH WAKTU
    dtArray = [0.1, 0.05, 0.025, 0.01, 0.005]   // Array berisi ukuran langkah
    max_errors = []                               // Array kosong untuk menyimpan error maksimum

// PERSIAPAN PLOT
    Buat figure dengan 2 subplot (ax1 di kiri, ax2 di kanan)

// PLOT SOLUSI ANALITIK (UNTUK REFERENSI)
    t_ref = buat array dari 0 sampai T dengan 500 titik
    x_ref = x0 * cos(omega * t_ref)               // Hitung solusi analitik
    Plot t_ref vs x_ref pada ax1 dengan garis hitam putus-putus

// LOOP UTAMA: ITERASI UNTUK SETIAP NILAI dt
    For each dt in dtArray:
        
        // HITUNG JUMLAH LANGKAH DAN ARRAY WAKTU
            N = T / dt (dibulatkan ke integer)
            t = buat array dari 0 sampai T dengan N+1 titik

        // INISIALISASI ARRAY SOLUSI NUMERIK
            x = array kosong berukuran N+1
            v = array kosong berukuran N+1
            x[0] = x0                                // Syarat awal posisi
            v[0] = v0                                // Syarat awal kecepatan

        // LOOP ITERASI NUMERIK: METODE FORWARD EULER
            For n from 0 to N-1:
                x[n+1] = x[n] + v[n] * dt            // Hitung posisi baru
                v[n+1] = v[n] - (omega^2) * x[n] * dt // Hitung kecepatan baru
            End For

        // PLOT SOLUSI NUMERIK (SUBPLOT KIRI)
            Plot t vs x pada ax1, beri label sesuai dt

        // HITUNG DAN SIMPAN ERROR MAKSIMUM
            x_true = x0 * cos(omega * t)             // Hitung solusi analitik pada waktu t yang sama
            max_err = nilai maksimum dari |x - x_true| untuk semua titik
            Tambahkan max_err ke dalam array max_errors

    End For

// PLOT UJI KONVERGENSI (SUBPLOT KANAN)
    Ubah max_errors menjadi array numpy untuk kemudahan
    Plot dtArray vs max_errors pada ax2 dengan skala log-log, tanda 'ro-' (lingkaran merah dengan garis)

// PLOT GARIS REFERENSI ORDE 1 (UNTUK PERBANDINGAN SLOPE)
    garis_referensi = max_errors[0] * (dtArray / dtArray[0])  // Skala linear berdasarkan error pertama
    Plot dtArray vs garis_referensi pada ax2 dengan garis biru putus-putus, label "Referensi Orde 1"

// PERAPIHAN DAN TAMPILAN PLOT
    Atur judul, label sumbu, legenda, dan grid untuk ax1
    Atur judul, label sumbu, legenda, dan grid untuk ax2
    Tampilkan plot (plt.show())
